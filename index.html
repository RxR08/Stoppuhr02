<script>

const display = document.getElementById("display");

const timeDisplay = document.getElementById("time");

const statusDisplay = document.getElementById("status");

const resetButton = document.getElementById("reset");

let running = false;

let startTime = 0;

let elapsed = 0;

let animationFrame;

function formatTime(ms) {

    const totalTenths = Math.floor(ms / 100);

    const minutes = Math.floor(totalTenths / 600);

    const seconds = Math.floor(totalTenths / 10) % 60;

    const tenths = totalTenths % 10;

    return (

        String(minutes).padStart(2, "0") + ":" +

        String(seconds).padStart(2, "0") + "." +

        tenths

    );

}

function updateDisplay() {

    if (!running) return;

    elapsed = performance.now() - startTime;

    timeDisplay.textContent = formatTime(elapsed);

    animationFrame = requestAnimationFrame(updateDisplay);

}

function toggleStopwatch() {

    if (!running) {

        // Stoppuhr starten

        running = true;

        startTime = performance.now() - elapsed;

        display.classList.add("running");

        statusDisplay.textContent = "LÄUFT";

        updateDisplay();

    } else {

        // Stoppuhr anhalten

        running = false;

        elapsed = performance.now() - startTime;

        cancelAnimationFrame(animationFrame);

        timeDisplay.textContent = formatTime(elapsed);

        display.classList.remove("running");

        statusDisplay.textContent = "GESTOPPT";

    }

}

// Gesamter Bildschirm startet und stoppt

display.addEventListener("pointerdown", (event) => {

    event.preventDefault();

    toggleStopwatch();

});

// Reset-Taste

resetButton.addEventListener("pointerdown", (event) => {

    event.preventDefault();

    event.stopPropagation();

    running = false;

    elapsed = 0;

    startTime = 0;

    cancelAnimationFrame(animationFrame);

    timeDisplay.textContent = "00:00.0";

    statusDisplay.textContent = "TIPPE ZUM STARTEN";

    display.classList.remove("running");

});

</script>
